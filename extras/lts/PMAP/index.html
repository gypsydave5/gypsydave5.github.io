private fun <T, R> Iterable<T>.pmap(
    maxConcurrency: Int,
    transform: (T) -> R,
): List<Result<PMapError, R>> {
    val semaphore = Semaphore(maxConcurrency + 1)
    val executor = Executors.newVirtualThreadPerTaskExecutor()

    try {
        val futures: List<Future<Result<PMapError, R>>> = map { item ->
            executor.submit<Result<PMapError, R>> {
                semaphore.acquire()
                try {
                    Success(transform(item))
                } catch (e: Throwable) {
                    Failure(PMapMError("Exception during pmap: $e", e))
                } finally {
                    semaphore.release()
                }
            }
        }

        return futures.map { it.get() }
    } finally {
        executor.shutdown()
    }
}
